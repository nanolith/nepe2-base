#Generation script for transformers_register
foreach(SRC IN LISTS NEPE2BASE_TRANSFORMER_SOURCES)
    file(READ "${SRC}" CONTENT)

    string(
        REGEX MATCHALL "TRANSFORMER_REGISTER\\s*\\([^,]+,[^)]+\\)"
        XFORM_REGISTRATIONS "${CONTENT}")

    foreach(CALL IN LISTS XFORM_REGISTRATIONS)
        if(CALL MATCHES "TRANSFORMER_REGISTER\\s*\\(([^,]+),([^)]+)\\)")
            string(STRIP "${CMAKE_MATCH_1}" NAME_ARG)
            string(STRIP "${CMAKE_MATCH_2}" FUNC_ARG)

            string(REPLACE "&" "" FUNC_NAME "${FUNC_ARG}")
            string(STRIP "${FUNC_NAME}" FUNC_NAME)

            string(APPEND DECLARATIONS
                "status ${FUNC_NAME}(\n    secure_buffer**, "
                "RCPR_SYM(allocator)*, const secure_buffer*,\n"
                "    const secure_buffer*, const metadata*);\n")
            string(APPEND REGISTRATIONS
                "    retval =\n"
                "        transformer_registry_register_inplace(\n"
                "            ${NAME_ARG},\n"
                "            ${FUNC_ARG});\n"
                "    if (STATUS_SUCCESS != retval)\n"
                "    {\n"
                "        return retval;\n"
                "    }\n\n")
        endif()
    endforeach()
endforeach()

set(OUTPUT_CONTENT "/* Auto-generated by CMake. Do not edit. */\n")
string(APPEND OUTPUT_CONTENT "#include <nepe2/transformer.h>\n\n")
string(APPEND OUTPUT_CONTENT "/* forward decls */\n")
string(APPEND OUTPUT_CONTENT "${DECLARATIONS}\n")
string(APPEND OUTPUT_CONTENT "status FN_DECL_MUST_CHECK ")
string(APPEND OUTPUT_CONTENT "transformers_register()\n")
string(APPEND OUTPUT_CONTENT "{\n")
string(APPEND OUTPUT_CONTENT "    status retval;\n\n")
string(APPEND OUTPUT_CONTENT "${REGISTRATIONS}")
string(APPEND OUTPUT_CONTENT "    return STATUS_SUCCESS;\n")
string(APPEND OUTPUT_CONTENT "}\n")

file(WRITE "${CMAKE_BINARY_DIR}/transformers_register.c" "${OUTPUT_CONTENT}")
